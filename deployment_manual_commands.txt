# Crear main.py del backend

from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import JSONResponse
import uvicorn
import os
from pathlib import Path

app = FastAPI(
    title="RITMO API",
    description="API para la plataforma RITMO",
    version="1.0.0"
)

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

@app.get("/health")
async def health_check():
    return {
        "status": "healthy",
        "service": "ritmo-backend",
        "version": "1.0.0"
    }

@app.get("/")
async def root():
    return {
        "message": "RITMO API is running",
        "version": "1.0.0",
        "docs": "/docs"
    }

@app.get("/info")
async def system_info():
    return {
        "python_version": "3.11+",
        "fastapi_version": "0.104.1",
        "environment": os.getenv("ENVIRONMENT", "production")
    }

if __name__ == "__main__":
    port = int(os.getenv("PORT", 8001))
    uvicorn.run(
        "main:app",
        host="0.0.0.0",
        port=port,
        reload=False
    )
EOF

# Crear configuración de Nginx
sudo tee /etc/nginx/sites-available/ritmo << 'EOF'
server {
    listen 80;
    server_name 46.62.165.154;

    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header X-Content-Type-Options "nosniff" always;

    location /api/ {
        proxy_pass http://127.0.0.1:8001/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location /health {
        proxy_pass http://127.0.0.1:8001/health;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
    }

    location /docs {
        proxy_pass http://127.0.0.1:8001/docs;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
    }

    location / {
        root /opt/ritmo/ritmo-frontend/dist;
        index index.html;
        try_files $uri $uri/ /index.html;
    }

    access_log /var/log/nginx/ritmo_access.log;
    error_log /var/log/nginx/ritmo_error.log;
}
EOF

# Habilitar sitio
sudo rm -f /etc/nginx/sites-enabled/default
sudo ln -sf /etc/nginx/sites-available/ritmo /etc/nginx/sites-enabled/

# Probar y reiniciar Nginx
sudo nginx -t
sudo systemctl reload nginx
sudo systemctl enable nginx

echo "✅ Nginx configurado"
