🐍 3. CREAR BACKEND (Copia bloque por bloque)
---------------------------------------------
cd /opt/ritmo && mkdir -p ritmo-backend && cd ritmo-backend

cat > requirements.txt << 'EOF'
fastapi==0.104.1
uvicorn[standard]==0.24.0
python-multipart==0.0.6
pydantic==2.5.0
python-dotenv==1.0.0
httpx==0.25.2
EOF

python3 -m venv venv && source venv/bin/activate && pip install --upgrade pip && pip install -r requirements.txt && deactivate

cat > main.py << 'EOF'
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
import uvicorn
import os

app = FastAPI(title="RITMO API", version="1.0.0")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

@app.get("/health")
async def health_check():
    return {"status": "healthy", "service": "ritmo-backend", "version": "1.0.0"}

@app.get("/")
async def root():
    return {"message": "RITMO API is running", "version": "1.0.0", "docs": "/docs"}

if __name__ == "__main__":
    port = int(os.getenv("PORT", 8001))
    uvicorn.run("main:app", host="0.0.0.0", port=port, reload=False)
EOF

🌐 6. CONFIGURAR NGINX (Copia todo)
-----------------------------------
sudo tee /etc/nginx/sites-available/ritmo << 'EOF'
server {
    listen 80;
    server_name 46.62.165.154;

    location /api/ {
        proxy_pass http://127.0.0.1:8001/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    location /health {
        proxy_pass http://127.0.0.1:8001/health;
        proxy_set_header Host $host;
    }

    location /docs {
        proxy_pass http://127.0.0.1:8001/docs;
        proxy_set_header Host $host;
    }

    location / {
        root /opt/ritmo/ritmo-frontend/dist;
        index index.html;
        try_files $uri $uri/ /index.html;
    }
}
EOF

sudo rm -f /etc/nginx/sites-enabled/default
sudo ln -sf /etc/nginx/sites-available/ritmo /etc/nginx/sites-enabled/
sudo nginx -t && sudo systemctl reload nginx && sudo systemctl enable nginx
